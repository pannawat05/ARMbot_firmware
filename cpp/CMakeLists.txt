cmake_minimum_required(VERSION 3.16)
project(pca9685_robot LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(pca9685_driver servo_controller.cpp)
target_compile_options(pca9685_driver PRIVATE -Wall -Wextra -Wpedantic)
add_executable(pca9685_servo_test main.cpp)
target_link_libraries(pca9685_servo_test PRIVATE pca9685_driver)
add_executable(pca9685_servo_health servo_health_test.cpp)
target_link_libraries(pca9685_servo_health PRIVATE pca9685_driver)
add_executable(servo_daemon servo_daemon.cpp)
target_link_libraries(servo_daemon PRIVATE pca9685_driver)
